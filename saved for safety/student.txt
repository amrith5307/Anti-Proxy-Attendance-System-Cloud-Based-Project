<script>
const idToken = () => sessionStorage.getItem("id_token");
document.getElementById("user-name").textContent =
  sessionStorage.getItem("user_name")||"";

function logout(){ sessionStorage.clear(); location.href="index.html"; }

let stream=null, scanning=false, scanDone=false;

function scanMsg(t,color="#94a3b8"){
  const el=document.getElementById("scan-msg");
  el.style.color=color; el.textContent=t;
}

async function startCamera(){
  scanDone=false;
  try{
    stream = await navigator.mediaDevices.getUserMedia({video:{facingMode:"environment"}});
    const v=document.getElementById("video");
    v.srcObject=stream;
    scanning=true;
    requestAnimationFrame(tick);
    scanMsg("Point camera at the QR code…","#94a3b8");
  }catch(e){ scanMsg("Camera error: "+e.message,"#f87171"); }
}

function stopCamera(){
  scanning=false;
  if(stream) stream.getTracks().forEach(t=>t.stop());
  stream=null;
  scanMsg("Camera stopped.","#94a3b8");
}

function tick(){
  if(!scanning) return;
  const v=document.getElementById("video");
  const c=document.getElementById("canvas");
  if(v.readyState===v.HAVE_ENOUGH_DATA){
    c.width=v.videoWidth; c.height=v.videoHeight;
    const ctx=c.getContext("2d");
    ctx.drawImage(v,0,0,c.width,c.height);
    const img=ctx.getImageData(0,0,c.width,c.height);
    const code=jsQR(img.data,img.width,img.height,{inversionAttempts:"dontInvert"});
    if(code && !scanDone){
      scanDone=true;
      stopCamera();
      handleQRData(code.data);
      return;
    }
  }
  requestAnimationFrame(tick);
}

async function handleQRData(raw){
  scanMsg("Processing…","#f59e0b");
  try{
    const { token } = JSON.parse(raw);
    const r = await fetch(API+"/attendance",{
      method:"POST",
      headers:{"Content-Type":"application/json","Authorization":idToken()},
      body: JSON.stringify({token})
    });
    const d = await r.json();
    if(!r.ok) throw new Error(d.error||"Failed");
    scanMsg(`✅ ${d.message} — ${d.subject}`,"#4ade80");
    loadHistory();
  }catch(e){ scanMsg("❌ "+e.message,"#f87171"); }
}

async function loadHistory(){
  try{
    const r = await fetch(API+"/reports",{
      headers:{"Authorization":idToken()}
    });
    const d = await r.json();
    const rows = d.attendance||[];
    document.getElementById("att-body").innerHTML =
      rows.map(a=>`<tr>
        <td>${a.subject||"—"}</td>
        <td>${new Date(a.marked_at*1000).toLocaleString()}</td>
      </tr>`).join("")||"<tr><td colspan=2 style='color:#64748b'>No records yet</td></tr>";
  }catch(e){ console.error(e); }
}

loadHistory();
